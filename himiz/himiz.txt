データベース名: himiz

create database himiz;

use himiz;

テーブル名: persons


テーブル定義

ID      id          int
名前    name        varchar(100)
性別    gender      char(2)
誕生日  birthday    date
出身    birthplace  varchar(100)

create table persons (
  id         int          primary key,
  name       varchar(100),
  gender     char(2),
  birthday   date,
  birthplace varchar(100)
);
