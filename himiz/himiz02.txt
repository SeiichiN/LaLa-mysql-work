データベース名: himiz02

create database himiz02;

use himiz02;

テーブル名: persons


テーブル定義

ID      id             int      プライマリキー 自動連番
名前    name           varchar(100)
性別    gender_id      int
誕生日  birthday       date
出身    prefecture_id  char(2)

create table persons (
  id            int    primary key auto_increment,
  name          varchar(100) not null,
  gender_id     int     not null,
  birthday      date    not null,
  prefecture_id char(2) not null
);


genders表の定義

ID    id   INT      PRIMARY KEY
名前  name VARCHAR(10)

CREATE TABLE genders (
  id   INT         PRIMARY KEY,
  name VARCHAR(10) NOT NULL
);
