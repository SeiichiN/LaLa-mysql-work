
sampleデータベース専用ユーザーの作成

CREATE USER 'sampleuser'@'localhost'
IDENTIFIED BY 'sampleuser';

sampleデータベースの中のすべての表に対する
権限を与える

GRANT ALL ON sample.* TO 'sampleuser'@'localhost';

sampleデータベースの作成

CREATE DATABASE sample;

--------------
文字列('') で指定: ユーザー名
                   ホスト名
                   パスワード

''を使ってはだめ : データベースの指定
                   テーブル名
                   カラム名
                   データ型
                   
                 
使用宣言

USE sample;

テーブル定義

CREATE TABLE employees (
  id       INT         AUTO_INCREMENT,
  name     VARCHAR(20) NOT NULL,
  age      INT         NOT NULL,
  birthday YEAR        NOT NULL,
  department_id CHAR(3),
  PRIMARY KEY (id)
);

CREATE TABLE departments (
  id   CHAR(3)     PRIMARY KEY,
  name VARCHAR(20) NOT NULL
);

=============================
 外部キー制約
=============================

employees表のdepartment_idの入力のとき、
departments表に登録されているIDしか
入力できないようにする。

ALTER TABLE employees  テーブル定義の修正
  ADD                  定義を追加
  CONSTRAINT fk_department_id  制約名をつける
  FOREIGN KEY (department_id) 制限をかけるカラムを指定
  REFERENCES departments (id)  参照する表・カラムを指定
  ON DELETE RESTRICT ON UPDATE RESTRICT
     何も書かなくても、これが適用される


外部キーにないデータの入力

INSERT INTO employees
  (name, age, birthday, department_id)
VALUES
  ('成田三樹夫', 38, 1935, '007');


部署表(departments表)を修正してみる

UPDATE departments
  SET id = '007'
  WHERE id = '002';

このコマンドは外部キー制約のため失敗する。

DELETE FROM departments
  WHERE id = '003';



データの修正

UPDATE employees
  SET name = '菅原文雄'
  WHERE id = 1;


データの削除

DELETE FROM employees
  WHERE id = 1;


全データの削除(auto_incrementは初期化されない)

DELETE FROM employees;


全データの削除

TRUNCATE employees;


データの表示・検索

SELECT * FROM employees
  WHERE name LIKE '%葉%';
